# Program 4: Function to print squares of numbers
print_squares <- function(n) {
cat("Squares of numbers from 1 to", n, ":\n")
for(i in 1:n) {
sq <- i^2
cat(i, "^2 =", sq, "\n")
}
}

# Using sapply (functional approach)
squares_vector <- function(vec) {
return(sapply(vec, function(x) x^2))
}

# Test
n <- 10
print_squares(n)

cat("\nUsing sapply:\n")
nums <- 1:5
result <- squares_vector(nums)
print(result)
