# Program to print even numbers from 1 to n

n <- as.integer(readline("Enter the value of n: "))

cat("Even numbers from 1 to", n, "are:\n")

for (i in 1:n) {

if (i %% 2 == 0) {
cat(i, " ")
}

}

cat("\n")

# %% is the modulus operator
# It gives the remainder after division
# If remainder is 0, the number is even

cat("Program completed successfully.")
